   532→   532→export function AiAgentsView() {
   533→   533→  const { user } = useAppStore();
   534→   534→  const [search, setSearch] = useState("");
   535→   535→  const [filterCategory, setFilterCategory] = useState<string>("all");
   536→   536→  const [togglingId, setTogglingId] = useState<string | null>(null);
   537→   537→  const [runningAgentId, setRunningAgentId] = useState<string | null>(null);
   538→   538→  const [runDialogAgent, setRunDialogAgent] = useState<AIAgent | null>(null);
   539→   539→  const [runTask, setRunTask] = useState("");
   540→   540→  const [runOutput, setRunOutput] = useState<string | null>(null);
   541→   541→  const [runError, setRunError] = useState<string | null>(null);
   542→   542→
   543→   543→  // Fetch real AI agents from /api/ai-agents
   544→   544→  const { data: apiData, isLoading, refetch } = useApiQuery<{ agents: any[] }>("/api/ai-agents", { enabled: true });
   545→   545→
   546→   546→  // Map DB agents to the view's expected shape.
   547→   547→  // If the DB has no agents yet, fall back to AGENT_DEFINITIONS seeded into the DB on first run.
   548→   548→  const agents: AIAgent[] = (apiData?.agents ?? []).map((a: any) => {
   549→   549→    const config = (() => { try { return JSON.parse(a.config || '{}'); } catch { return {}; } })();
   550→   550→    return {
   551→   551→      id: a.id,
   552→   552→      name: a.name,
   553→   553→      description: a.description || '',
   554→   554→      category: config.category || 'Strategy',
   555→   555→      type: a.type,
   556→   556→      enabled: a.isEnabled,
   557→   557→      lastActive: a.updatedAt || a.createdAt,
   558→   558→      tasksCompleted: config.tasksCompleted || 0,
   559→   559→    };
   560→   560→  });
   561→   561→
   562→   562→  // Mutation for toggling agents — calls /api/ai-agents PATCH with { id, isEnabled }
   563→   563→  const toggleMutation = useApiMutation({
   564→   564→    method: "PATCH",
   565→   565→    path: "/api/ai-agents",
   566→   566→    onSuccess: () => { refetch(); },
   567→   567→    onError: () => { toast.error("Failed to update agent status"); refetch(); },
   568→   568→  });
   569→   569→
   570→   570→  const handleToggle = useCallback(
   571→   571→    async (agentId: string, enabled: boolean) => {
   572→   572→      setTogglingId(agentId);
   573→   573→      try {
   574→   574→        toggleMutation.mutate({ id: agentId, isEnabled: enabled });
   575→   575→        toast.success(`${agents.find((a) => a.id === agentId)?.name || 'Agent'} ${enabled ? "enabled" : "disabled"}`);
   576→   576→      } catch {
   577→   577→        toast.error("Failed to update agent status");
   578→   578→      } finally {
   579→   579→        setTogglingId(null);
   580→   580→      }
   581→   581→    },
   582→   582→    [toggleMutation, agents]
   583→   583→  );
   584→   584→
   585→   585→  const handleRunAgent = useCallback(
   586→   586→    async (agent: AIAgent) => {
   587→   587→      setRunDialogAgent(agent);
   588→   588→      setRunTask("");

   590→   590→      setRunError(null);
   591→   591→    },
   592→   592→    [],
   593→   593→  );
   594→   594→
   595→   595→  const handleExecuteAgent = useCallback(async () => {

   598→   598→    setRunOutput(null);
   599→   599→    setRunError(null);
   600→   600→    try {

   602→   602→      const res = await fetch(`/api/ai/agents/${runDialogAgent.id}/run`, {
   603→   603→        method: "POST",
   604→   604→        headers: { "Content-Type": "application/json", ...(token ? { Authorization: `Bearer ${token}` } : {}) },
   605→   605→        body: JSON.stringify({ task: runTask }),
   606→   606→      });
   607→   607→      if (!res.ok) {
   608→   608→        const err = await res.json().catch(() => ({}));
   609→   609→        throw new Error(err.error || `HTTP ${res.status}`);
   610→   610→      }
   611→   611→      const json = await res.json();
   612→   612→      const result = json.data || json;
   613→   613→      if (result.success === false) {
   614→   614→        setRunError(result.error || "Agent failed to produce output");
   615→   615→        toast.error(`Agent failed: ${result.error || "unknown error"}`);
   616→   616→      } else {
   617→   617→        setRunOutput(result.output);
   618→   618→        toast.success(`${runDialogAgent.name} completed in ${result.durationMs}ms`);
   619→   619→        refetch(); // refresh task counts
   620→   620→      }
   621→   621→    } catch (err) {

   624→   624→      toast.error(msg);
   625→   625→    } finally {
   626→   626→      setRunningAgentId(null);

   645→   645→  });
   646→   646→
   647→   647→  if (isLoading) return <AgentsSkeleton />;
   648→   648→
   649→   649→  return (

   651→   651→      className="space-y-6"
   652→   652→      variants={containerVariants}
   653→   653→      initial="hidden"

   655→   655→    >

   661→   661→              <Bot className="h-5 w-5" />
