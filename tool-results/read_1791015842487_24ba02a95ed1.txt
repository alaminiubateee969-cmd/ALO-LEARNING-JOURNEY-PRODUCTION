   223→/* --------------------------- helpers --------------------------- */
   224→
   225→function maskRecipient(recipient: string): string {
   226→  if (!recipient) return "—";
   227→  // email: a***@domain.com
   228→  if (recipient.includes("@")) {
   229→    const [u, d] = recipient.split("@");
   230→    if (!u || !d) return recipient;
   231→    const head = u.slice(0, Math.min(2, u.length));
   232→    return `${head}${"•".repeat(Math.max(2, u.length - 2))}@${d}`;
   233→  }
   234→  // phone: keep country code & last 2
   235→  const trimmed = recipient.replace(/\s|-/g, "");
   236→  if (trimmed.length <= 4) return "•".repeat(trimmed.length);
   237→  const head = trimmed.startsWith("+") ? "+" : "";
   238→  const body = trimmed.startsWith("+") ? trimmed.slice(1) : trimmed;
   239→  const last2 = body.slice(-2);
   240→  return `${head}${"•".repeat(Math.max(2, body.length - 2))}${last2}`;
   241→}
   242→
   243→function formatMMSS(totalSeconds: number): string {
   244→  const s = Math.max(0, Math.floor(totalSeconds));
   245→  const m = Math.floor(s / 60);
   246→  const r = s % 60;
   247→  return `${String(m).padStart(2, "0")}:${String(r).padStart(2, "0")}`;
   248→}
   249→
   250→function formatTime(iso: string): string {
   251→  if (!iso) return "—";
   252→  try {
   253→    return new Date(iso).toLocaleString(undefined, {
   254→      month: "short",
   255→      day: "2-digit",
   256→      hour: "2-digit",
   257→      minute: "2-digit",
   258→    });
   259→  } catch {
   260→    return iso;
   261→  }
   262→}
