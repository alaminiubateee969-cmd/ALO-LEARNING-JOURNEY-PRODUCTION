   532→export function AiAgentsView() {
   533→  const { user } = useAppStore();
   534→  const [search, setSearch] = useState("");
   535→  const [filterCategory, setFilterCategory] = useState<string>("all");
   536→  const [togglingId, setTogglingId] = useState<string | null>(null);
   537→  const [runningAgentId, setRunningAgentId] = useState<string | null>(null);
   538→  const [runDialogAgent, setRunDialogAgent] = useState<AIAgent | null>(null);
   539→  const [runTask, setRunTask] = useState("");
   540→  const [runOutput, setRunOutput] = useState<string | null>(null);
   541→  const [runError, setRunError] = useState<string | null>(null);
   542→
   543→  // Fetch real AI agents from /api/ai-agents
   544→  const { data: apiData, isLoading, refetch } = useApiQuery<{ agents: any[] }>("/api/ai-agents", { enabled: true });
   545→
   546→  // Map DB agents to the view's expected shape.
   547→  // If the DB has no agents yet, fall back to AGENT_DEFINITIONS seeded into the DB on first run.
   548→  const agents: AIAgent[] = (apiData?.agents ?? []).map((a: any) => {
   549→    const config = (() => { try { return JSON.parse(a.config || '{}'); } catch { return {}; } })();
   550→    return {
   551→      id: a.id,
   552→      name: a.name,
   553→      description: a.description || '',
   554→      category: config.category || 'Strategy',
   555→      type: a.type,
   556→      enabled: a.isEnabled,
   557→      lastActive: a.updatedAt || a.createdAt,
   558→      tasksCompleted: config.tasksCompleted || 0,
   559→    };
   560→  });
   561→
   562→  // Mutation for toggling agents — calls /api/ai-agents PATCH with { id, isEnabled }
   563→  const toggleMutation = useApiMutation({
   564→    method: "PATCH",
   565→    path: "/api/ai-agents",
   566→    onSuccess: () => { refetch(); },
   567→    onError: () => { toast.error("Failed to update agent status"); refetch(); },
   568→  });
   569→
   570→  const handleToggle = useCallback(
   571→    async (agentId: string, enabled: boolean) => {
   572→      setTogglingId(agentId);
   573→      try {
   574→        toggleMutation.mutate({ id: agentId, isEnabled: enabled });
   575→        toast.success(`${agents.find((a) => a.id === agentId)?.name || 'Agent'} ${enabled ? "enabled" : "disabled"}`);
   576→      } catch {
   577→        toast.error("Failed to update agent status");
   578→      } finally {
   579→        setTogglingId(null);
   580→      }
   581→    },
   582→    [toggleMutation, agents]
   583→  );
   584→
   585→  const handleRunAgent = useCallback(
   586→    async (agent: AIAgent) => {
   587→      setRunDialogAgent(agent);
   588→      setRunTask("");
   589→      setRunOutput(null);
   590→      setRunError(null);
   591→    },
   592→    [],
   593→  );
   594→
   595→  const handleExecuteAgent = useCallback(async () => {
   596→    if (!runDialogAgent || !runTask.trim()) return;
   597→    setRunningAgentId(runDialogAgent.id);
   598→    setRunOutput(null);
   599→    setRunError(null);
   600→    try {
   601→      const token = localStorage.getItem("mos_token");
   602→      const res = await fetch(`/api/ai/agents/${runDialogAgent.id}/run`, {
   603→        method: "POST",
   604→        headers: { "Content-Type": "application/json", ...(token ? { Authorization: `Bearer ${token}` } : {}) },
   605→        body: JSON.stringify({ task: runTask }),
   606→      });
   607→      if (!res.ok) {
   608→        const err = await res.json().catch(() => ({}));
   609→        throw new Error(err.error || `HTTP ${res.status}`);
   610→      }
   611→      const json = await res.json();
   612→      const result = json.data || json;
   613→      if (result.success === false) {
   614→        setRunError(result.error || "Agent failed to produce output");
   615→        toast.error(`Agent failed: ${result.error || "unknown error"}`);
   616→      } else {
   617→        setRunOutput(result.output);
   618→        toast.success(`${runDialogAgent.name} completed in ${result.durationMs}ms`);
   619→        refetch(); // refresh task counts
   620→      }
   621→    } catch (err) {
   622→      const msg = err instanceof Error ? err.message : "Failed to run agent";
   623→      setRunError(msg);
   624→      toast.error(msg);
   625→    } finally {
   626→      setRunningAgentId(null);
   627→    }
   628→  }, [runDialogAgent, runTask, refetch]);
   629→
   630→  // Calculate stats from real agent list
   631→  const stats = {
   632→    totalAgents: agents.length,
   633→    activeAgents: agents.filter((a) => a.enabled).length,
   634→    categories: [...new Set(agents.map((a) => a.category))],
   635→  };
   636→
   637→  // Filter agents
   638→  const filteredAgents = agents.filter((agent) => {
   639→    const matchesSearch =
   640→      agent.name.toLowerCase().includes(search.toLowerCase()) ||
   641→      agent.description.toLowerCase().includes(search.toLowerCase()) ||
   642→      agent.category.toLowerCase().includes(search.toLowerCase());
   643→    const matchesCategory = filterCategory === "all" || agent.category === filterCategory;
   644→    return matchesSearch && matchesCategory;
   645→  });
   646→
   647→  if (isLoading) return <AgentsSkeleton />;
   648→
   649→  return (
   650→    <motion.div
   651→      className="space-y-6"
   652→      variants={containerVariants}
   653→      initial="hidden"
   654→      animate="visible"
   655→    >
   656→      {/* Header */}
   657→      <motion.div variants={itemVariants} className="flex items-center justify-between flex-wrap gap-3">
   658→        <div>
   659→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   660→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-gradient-to-br from-orange-500 to-blue-500 text-white shadow-lg shadow-orange-500/25">
   661→              <Bot className="h-5 w-5" />
   662→            </div>
   663→            AI Agent Control Panel
   664→          </h1>
   665→          <p className="text-muted-foreground text-sm mt-1">
   666→            Manage and control your AI-powered marketing agents
   667→          </p>
   668→        </div>
   669→        <div className="flex items-center gap-2">
   670→          <Badge variant="outline" className="bg-emerald-500/10 text-emerald-500 border-emerald-500/20">
   671→            <div className="h-1.5 w-1.5 rounded-full bg-emerald-500 mr-1.5" />
   672→            {stats.activeAgents} Active
   673→          </Badge>
   674→        </div>
   675→      </motion.div>
   676→
   677→      {/* Stats */}
   678→      <motion.div variants={itemVariants}>
   679→        <StatsRow stats={stats} />
   680→      </motion.div>
   681→
   682→      {/* Filters */}
   683→      <motion.div variants={itemVariants} className="flex items-center gap-3 flex-wrap">
   684→        <div className="relative flex-1 max-w-sm">
   685→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   686→          <Input
   687→            placeholder="Search agents..."
   688→            value={search}
   689→            onChange={(e) => setSearch(e.target.value)}
   690→            className="pl-9"
   691→          />
   692→        </div>
   693→        <div className="flex items-center gap-1 rounded-lg bg-muted/50 p-1 flex-wrap">
   694→          <button
   695→            onClick={() => setFilterCategory("all")}
   696→            className={`px-3 py-1.5 text-xs font-medium rounded-md transition-all ${
   697→              filterCategory === "all"
   698→                ? "bg-background text-foreground shadow-sm"
   699→                : "text-muted-foreground hover:text-foreground"
   700→            }`}
   701→          >
   702→            All
   703→          </button>
   704→          {stats.categories.map((cat) => {
   705→            const CatIcon = CATEGORY_ICONS[cat] || Bot;
   706→            return (
   707→              <button
   708→                key={cat}
   709→                onClick={() => setFilterCategory(cat)}
   710→                className={`px-3 py-1.5 text-xs font-medium rounded-md transition-all flex items-center gap-1 ${
   711→                  filterCategory === cat
   712→                    ? "bg-background text-foreground shadow-sm"
   713→                    : "text-muted-foreground hover:text-foreground"
   714→                }`}
   715→              >
   716→                <CatIcon className="h-3 w-3" />
   717→                {cat}
   718→              </button>
   719→            );
   720→          })}
   721→        </div>
   722→      </motion.div>
   723→
   724→      {/* Agent Grid */}
   725→      <motion.div
   726→        variants={containerVariants}
   727→        className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3"
   728→      >
   729→        {filteredAgents.map((agent) => (
   730→          <AgentCard
   731→            key={agent.id}
   732→            agent={agent}
   733→            onToggle={handleToggle}
   734→            isToggling={togglingId === agent.id}
   735→            onRun={handleRunAgent}
   736→            isRunning={runningAgentId === agent.id}
   737→          />
   738→        ))}
   739→      </motion.div>
   740→
   741→      {filteredAgents.length === 0 && (
   742→        <div className="flex flex-col items-center justify-center py-12 text-muted-foreground">
   743→          <Bot className="h-10 w-10 opacity-30 mb-2" />
   744→          <p className="text-sm">No agents found matching your filters</p>
   745→        </div>
   746→      )}
   747→
   748→      {/* Run Agent Dialog */}
   749→      <Dialog open={!!runDialogAgent} onOpenChange={(v) => !v && setRunDialogAgent(null)}>
   750→        <DialogContent className="sm:max-w-2xl bg-white/95 dark:bg-slate-900/95 backdrop-blur-xl border-white/20 dark:border-white/10 max-h-[90vh] overflow-y-auto">
   751→          <DialogHeader>
   752→            <DialogTitle className="flex items-center gap-2">
   753→              <span className="text-lg">{runDialogAgent?.icon}</span>
   754→              Run Agent: {runDialogAgent?.name}
   755→            </DialogTitle>
   756→            <DialogDescription>
   757→              Type a task for the {runDialogAgent?.name} agent to execute via the LLM. The agent will save its output to the activity feed + audit log.
   758→            </DialogDescription>
   759→          </DialogHeader>
   760→
   761→          <div className="space-y-3">
   762→            <div className="space-y-1.5">
   763→              <label className="text-xs font-medium">Task description</label>
   764→              <textarea
   765→                value={runTask}
   766→                onChange={(e) => setRunTask(e.target.value)}
   767→                placeholder="e.g. Score this lead: Sarah Chen from TechVault Inc, source LinkedIn, currently 65"
   768→                disabled={runningAgentId === runDialogAgent?.id}
   769→                className="w-full min-h-[80px] rounded-md border border-border/50 bg-white/80 dark:bg-slate-800/80 px-3 py-2 text-sm placeholder:text-muted-foreground focus:outline-none focus:ring-2 focus:ring-orange-500/50"
   770→                rows={3}
   771→              />
   772→            </div>
   773→
   774→            {runningAgentId === runDialogAgent?.id && (
   775→              <div className="flex items-center gap-2 text-xs text-orange-600 dark:text-orange-400">
   776→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   777→                Calling LLM... (this may take 5-15 seconds)
   778→              </div>
   779→            )}
   780→
   781→            {runError && (
   782→              <div className="rounded-md border border-red-500/30 bg-red-500/5 p-3 text-xs text-red-700 dark:text-red-300">
   783→                <strong>Error:</strong> {runError}
   784→              </div>
   785→            )}
   786→
   787→            {runOutput && (
   788→              <div className="space-y-1.5">
   789→                <label className="text-xs font-medium">Agent output</label>
   790→                <pre className="rounded-md border border-emerald-500/30 bg-emerald-500/5 p-3 text-xs whitespace-pre-wrap font-mono max-h-[300px] overflow-y-auto">
   791→                  {runOutput}
   792→                </pre>
   793→              </div>
   794→            )}
   795→          </div>
   796→
   797→          <DialogFooter>
   798→            <Button variant="ghost" onClick={() => setRunDialogAgent(null)} disabled={!!runningAgentId}>
   799→              Close
   800→            </Button>
   801→            <Button
   802→              onClick={handleExecuteAgent}
   803→              disabled={!runTask.trim() || runningAgentId === runDialogAgent?.id}
   804→              className="bg-gradient-to-r from-orange-500 to-orange-600 hover:from-orange-600 hover:to-orange-700 text-white"
   805→            >
   806→              {runningAgentId === runDialogAgent?.id ? (
   807→                <><Loader2 className="h-4 w-4 mr-1.5 animate-spin" /> Running...</>
   808→              ) : (
   809→                <><Send className="h-4 w-4 mr-1.5" /> Run Agent</>
   810→              )}
   811→            </Button>
   812→          </DialogFooter>
   813→        </DialogContent>
   814→      </Dialog>
   815→    </motion.div>
   816→  );
   817→}
   818→
   819→export default AiAgentsView;
