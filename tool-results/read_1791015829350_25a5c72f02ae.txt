   111→// ---------- Helpers ----------
   112→function formatBytes(bytes: number): string {
   113→  if (!bytes) return "0 B";
   114→  const units = ["B", "KB", "MB"];
   115→  const i = Math.min(units.length - 1, Math.floor(Math.log(bytes) / Math.log(1024)));
   116→  return `${(bytes / Math.pow(1024, i)).toFixed(i === 0 ? 0 : 1)} ${units[i]}`;
   117→}
   118→
   119→function timeAgo(iso: string): string {
   120→  const diff = Date.now() - new Date(iso).getTime();
   121→  const m = Math.floor(diff / 60000);
   122→  if (m < 1) return "just now";
   123→  if (m < 60) return `${m}m ago`;
   124→  const h = Math.floor(m / 60);
   125→  if (h < 24) return `${h}h ago`;
   126→  return `${Math.floor(h / 24)}d ago`;
   127→}
